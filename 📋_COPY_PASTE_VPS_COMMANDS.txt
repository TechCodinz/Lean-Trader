════════════════════════════════════════════════════════════════════════════════
STEP 6: STOP OLD BOT (If running)
════════════════════════════════════════════════════════════════════════════════

# Find running bot process
ps aux | grep python | grep -i bot

# Kill it (replace PID with actual number)
kill <PID>

# Or kill all python bots
pkill -9 -f "python.*BOT"

# Verify nothing running
ps aux | grep python | grep -i bot

════════════════════════════════════════════════════════════════════════════════
STEP 7: START NEW BOT
════════════════════════════════════════════════════════════════════════════════

# Method 1: Docker (Recommended)
docker-compose down
docker-compose build
docker-compose up -d
docker-compose logs -f trading-bot

# Method 2: Direct Python
nohup python3 AUTOMATED_ORCHESTRATOR_ENHANCED.py > logs/bot.log 2>&1 &
echo $! > bot.pid
tail -f logs/bot.log

# Method 3: Systemd Service
sudo systemctl restart apex-bot
sudo journalctl -u apex-bot -f
