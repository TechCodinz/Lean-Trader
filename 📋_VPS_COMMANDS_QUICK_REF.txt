# Or if using nohup.out:

tail -50 apex.log

# Last 100 lines:
tail -100 apex.log

════════════════════════════════════════════════════════════════════════════════
🔎 CHECK FOR RECENT TRADES
════════════════════════════════════════════════════════════════════════════════

# Recent trades:
grep -i "trade\|signal\|executed" apex.log | tail -20

# Recent profits:
grep -i "profit\|win\|loss" apex.log | tail -20

# Recent signals:
grep -i "signal.*confidence" apex.log | tail -20

════════════════════════════════════════════════════════════════════════════════
❌ CHECK FOR ERRORS
════════════════════════════════════════════════════════════════════════════════

# Recent errors:
grep -i "error\|exception\|failed" apex.log | tail -20

# Critical errors:
grep -iE "critical|fatal|traceback" apex.log | tail -20

# Count errors:
grep -i "error" apex.log | wc -l

════════════════════════════════════════════════════════════════════════════════
🗄️ CHECK EVOLUTION DATABASE
════════════════════════════════════════════════════════════════════════════════

# Check if database exists:
ls -lh evolution_engine.db

# Count rows:
sqlite3 evolution_engine.db "SELECT COUNT(*) FROM model_evolution;"

# Get max generation:
sqlite3 evolution_engine.db "SELECT MAX(generation) FROM model_evolution;"

# Get recent entries:
sqlite3 evolution_engine.db "SELECT generation, model_type, performance_score FROM model_evolution ORDER BY generation DESC LIMIT 10;"
