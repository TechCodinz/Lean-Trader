=============================================================================
⚡ CHECK PROFIT MONITORING STATUS
=============================================================================

cd ~/bot

echo "1️⃣ HOW MANY TRADES ARE BEING MONITORED?"
grep -c "Profit monitoring started" bot.log

echo ""
echo "2️⃣ WHAT ARE THE PROFIT TARGETS?"
grep "TP1:\|TP.*Level\|Target.*profit" bot.log | tail -10

echo ""
echo "3️⃣ SHOW MONITORING DETAILS:"
grep "Starting.*profit monitoring\|Entry:\|TP1:\|SL:\|Max hold:" bot.log | tail -30

echo ""
echo "4️⃣ IS MONITORING LOOP CHECKING PRICES?"
grep "P&L\|current.*price\|Held for" bot.log | tail -20

echo ""
echo "5️⃣ WHAT'S THE HOLD TIME?"
grep "Max hold:" bot.log | tail -10

echo ""
echo "6️⃣ ANY MONITORING ERRORS?"
grep "Monitoring error\|monitoring.*error" bot.log

echo ""
echo "7️⃣ SHOW LAST 50 LINES (FULL CONTEXT):"
tail -50 bot.log
